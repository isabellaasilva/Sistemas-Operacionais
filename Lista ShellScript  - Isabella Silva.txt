1) #!/bin/bash 
clear 
read -p "Informe n1: " n1
read -p "Informe n2: " n2

echo "3)Multiplicação" 
echo "4)Divisão" 
echo "5)Outro" 
read -p "Digite a operação escolhida: " x

1) echo "Soma" 
soma=$(expr $n1 + $n2) 
echo "O Resultado da soma é '$soma'" 
;; 
2)echo "Subtração" 
soma=$(expr $n1 - $n2) 
echo "O Resultado da subtração é '$soma'"
;;
3)echo "Multiplicação" 
soma=$(expr $n1 \* $n2) 
echo "O Resultado da multiplicação é '$soma'" 
;;
4)echo "Divisão" 
soma=$(expr $n1 / $n2) 
echo "O Resultado da divisão é '$soma'" 
;; 
*)echo "Outro" 
esac

10) #!/bin/bash 
clear 
echo "Digite o nome do usuário:"
read usuario
linha=$(grep "^$usuario:" /etc/passwd)
if [ -z "$linha" ]; then
    echo "Usuário não encontrado."
else
    home=$(echo "$linha" | cut -d: -f6)
    echo "Nome do usuário: $usuario"
    echo "Diretório de trabalho (home): $home"
    if [ -d "$home" ]; then
        espaco=$(du -sh "$home" 2>/dev/null | cut -f1)
        echo "Espaço utilizado no disco: $espaco"
    else
        echo "Não foi possível acessar o diretório $home (pode não existir ou faltar permissão)."
    fi
fi
